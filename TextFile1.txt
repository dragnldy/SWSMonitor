To generate a dynamic JSON feed of your Google Drive images automatically, you can use Google Apps Script deployed as a Web App. [1, 2, 3] 
This method reads a specific Google Drive folder, extracts the valid direct image links, and outputs them as a clean JSON API endpoint that your HTML carousel can fetch.
------------------------------
## Step 1: Create the Google Apps Script

   1. Go to Google Apps Script.
   2. Click New Project.
   3. Clear any default code in Code.gs and paste the following script: [4, 5] 

function doGet() {
  // 1. REPLACE WITH YOUR ACTUAL GOOGLE DRIVE FOLDER ID
  const folderId = "YOUR_GOOGLE_DRIVE_FOLDER_ID"; 
  
  try {
    const folder = DriveApp.getFolderById(folderId);
    const files = folder.getFiles();
    const imagesList = [];
    
    // 2. Loop through all files in the target folder
    while (files.hasNext()) {
      const file = files.next();
      const mimeType = file.getMimeType();
      
      // Filter for standard web image formats
      if (mimeType.startsWith("image/")) {
        imagesList.push({
          id: file.getId(),
          name: file.getName(),
          // Direct web URL format for Google Drive files
          url: "https://google.com" + file.getId(),
          // Thumbnail URL format (faster loading for high-res images)
          thumbnailUrl: "https://google.com" + file.getId() + "&sz=w1000"
        });
      }
    }
    
    // 3. Construct and return the JSON response with CORS headers allowed
    const output = ContentService.createTextOutput(JSON.stringify(imagesList));
    output.setMimeType(ContentService.MimeType.JSON);
    return output;
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ error: error.toString() }))
                         .setMimeType(ContentService.MimeType.JSON);
  }
}

Would you like help with enabling virtualization (lazy loading) for this code to prevent performance lag if your Google Drive folder contains hundreds of images?

Here is the updated code implementing high-performance virtualized lazy loading.
Instead of generating DOM nodes and loading image payloads for all files at once, this version creates empty structural wrappers ("shells") for the slides. It uses the browser's native Intersection Observer API to dynamically inject the src attribute and fetch the Google Drive image asset only when that specific slide is about to enter the viewport. [1, 2, 3]

<script>
  // paste YOUR COPIED APPS SCRIPT WEB APP URL here
  const API_URL = "https://google.com";
  
  const track = document.getElementById('track');
  const statusMsg = document.getElementById('status');
  let currentIndex = 0;
  let totalSlides = 0;
  let imageLoaderObserver;

  // Initialize Intersection Observer for Virtual Lazy Loading
  function initLazyLoading() {
    imageLoaderObserver = new IntersectionObserver((entries, observer) => {
      entries.forEach(entry => {
        // If slide structural shell enters viewport (or is near it)
        if (entry.isIntersecting) {
          const slide = entry.target;
          const img = slide.querySelector('img');
          const spinner = slide.querySelector('.spinner');
          
          if (img && img.dataset.src) {
            // Move dataset source into live DOM asset source string
            img.src = img.dataset.src;
            img.removeAttribute('data-src');
            
            img.onload = () => {
              img.classList.add('loaded');
              if (spinner) spinner.remove(); // Clean memory and DOM tree
            };
          }
          // Stop observing this element since asset is successfully requested
          observer.unobserve(slide);
        }
      });
    }, {
      root: track.parentElement, // Observe relative bounding box of viewport
      rootMargin: "0px 100px 0px 100px" // Pre-load upcoming images 100px before they slide into view
    });
  }

  async function loadCarouselImages() {
    try {
      const response = await fetch(API_URL);
      const images = await response.json();
      
      if (images.error || images.length === 0) {
        statusMsg.textContent = "No assets found or restricted access permissions.";
        return;
      }

      // Flush status message from container track layout
      track.innerHTML = '';
      totalSlides = images.length;
      initLazyLoading();

      // Step 2: Build virtualized slide structures cleanly without requesting asset payloads
      images.forEach((imgData) => {
        const slide = document.createElement('div');
        slide.className = 'carousel-slide';
        
        // Loader animation node
        const spinner = document.createElement('div');
        spinner.className = 'spinner';
        
        // Blank image blueprint pointing to data-src instead of real src
        const img = document.createElement('img');
        img.setAttribute('data-src', imgData.thumbnailUrl); 
        img.alt = imgData.name;
        
        slide.appendChild(spinner);
        slide.appendChild(img);
        track.appendChild(slide);
        
        // Target individual empty wrapper to the lazy loader queue
        imageLoaderObserver.observe(slide);
      });

    } catch (err) {
      console.error(err);
      statusMsg.textContent = "Network initialization failure.";
    }
  }

  function moveSlide(direction) {
    if (totalSlides === 0) return;
    currentIndex += direction;
    
    // Looping structural boundaries
    if (currentIndex < 0) currentIndex = totalSlides - 1;
    if (currentIndex >= totalSlides) currentIndex = 0;
    
    // Move track window coordinates via hardware-accelerated transform matrix
    track.style.transform = `translateX(-${currentIndex * 100}%)`;
  }

  // Execute structural initialization immediately
  loadCarouselImages();
</script>

## Architecture Highlights

* Zero Network Bloat: If your drive folder contains 500 images, your phone or desktop browser only requests data pipelines for the 2–3 images currently sitting inside or immediately adjacent to your container framework view.
* rootMargin Optimization: The observer features a configured buffer boundary of 100px. This triggers the background download of the upcoming image slightly before the user clicks the navigation arrows, hiding loading lag.
